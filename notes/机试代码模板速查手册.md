### 2.2 string
```cpp
string s, t;
s.size(); s.length();          // 长度
s.substr(pos, cnt);            // 切 [pos, pos+cnt)，cnt 是长度
s.substr(pos);                 // 切到结尾
s.find(c);                     // 找字符/串，返回下标，找不到返回 string::npos
s.find(c, pos);                // 从 pos 起找；判存在：if (s.find(x) != string::npos)
count(s.begin(), s.end(), 'a');// 统计字符出现次数（<algorithm>）；带条件用 count_if + lambda
// 子串计数（无现成函数，靠 find 循环，p 用 size_t）：
//   for (size_t p = s.find(t); p != string::npos; p = s.find(t, p + t.size())) cnt++;
// ⚠️ 统计"每个字符各几次"别对每个字符都调一次 count（O(26n)）→ 桶计数一遍扫完（§3.9）
s += t;  s == t;  s < t;       // 拼接/比较（字典序）
s.append(t);                   // 等价 s += t（append 没有单字符版！单字符用 +=）
s.append(t, pos, len);         // 追加 t 的子串 [pos,pos+len)（+= 做不到，省临时对象）
s.append(buf, n);              // 追加 char 数组前 n 个字符（二进制安全，可含 '\0'）
s.append(n, ch);               // 追加 n 个 ch
// ⚠️ pos 越界抛 out_of_range；len 超长只截断到源串结尾
s.erase(pos, cnt);  s.insert(pos, t);
s.erase(pos);                  // 从 pos 删到末尾
s.insert(pos, cnt, ch);        // 在 pos【之前】插入 cnt 个 ch（原 pos 处字符整体后移）
// 想插在某个字符【之后】：pos = s.find(c) + 1；pos == s.size() 即末尾追加；pos > size() 抛 out_of_range
s.replace(pos, cnt, t);        // 把 [pos, pos+cnt) 替换为 t
reverse(s.begin(), s.end());   // 反转（<algorithm>）
printf("%s\n", s.c_str());     // 转 char* 给 C 函数用（c_str() 保证 '\0'）
stoi(s); stoll(s); stod(s);    // 字符串→数字（吃前段；全非法抛 invalid_argument、越界抛 out_of_range）
string s = to_string(x);                  // 数字→字符串（int/LL/unsigned/double…都有重载）
// ⚠️ to_string 三坑：char→"65"(ASCII 提升 int)；bool→"1"/"0"
//    double 恒 6 位小数(100.0→"100.000000")→ 要格式用 printf/ostringstream
// ⚠️ 大数加 LL：to_string(1234567890123LL)；单字符字符串用 string(1,ch) 或 s.push_back(ch)
```


**priority_queue 完整定义（oi-wiki 模板）**：
```cpp
priority_queue<TypeName> q;                        // 数据类型 TypeName
priority_queue<TypeName, Container> q;             // 指定底层容器（默认 vector）
priority_queue<TypeName, Container, Compare> q;    // 指定比较类型
// 比较类型 less<TypeName>：top() 返回最大值（大根堆，默认）
// 比较类型 greater<TypeName>：top() 返回最小值（小根堆）
// 注意：不可跳过 Container 直接传入 Compare（三个参数要么都写要么都不写）

// C++11 起可用 lambda 自定义 Compare（须作为构造参数代入）：
auto cmp = [](const pair<int, int>& l, const pair<int, int>& r) {
    return l.second < r.second;   // 按 second 大根堆
};
priority_queue<pair<int, int>, vector<pair<int, int>>, decltype(cmp)> pq(cmp);

// 小根堆实测（oi-wiki 示例）：
priority_queue<int, vector<int>, greater<int>> q3;   // q3 为小根堆
for (int i = 1; i <= 5; i++) q3.push(i);
q3.top();   // 1（最小值在堆顶）；q1 大根堆同理 top()=5
```

**⚠️ priority_queue 没有 count / find / erase / 迭代器**（它是容器适配器，只暴露 top/empty/size/push/pop/swap）
编译实测：`pq.count(x)`、`pq.begin()`、`pq.erase(top())` 都会报 `has no member named ...`。要按值计数/查找/删除，选一个：

```cpp
// ① 推荐：另开一张计数表，push/pop 时同步维护（O(1)）
unordered_map<int,int> cnt;
auto push = [&](int x){ pq.push(x); cnt[x]++; };
auto pop  = [&](){ int t = pq.top(); pq.pop(); if(--cnt[t] == 0) cnt.erase(t); return t; };
cnt[5];                                   // 直接问"5 有几个"

// ② 需要"有序 + 计数 + 删任意元素" → 换 multiset（count O(log n + 个数)）
multiset<int> ms;  ms.count(5);  ms.erase(ms.find(5));  *ms.begin();

// ③ 要自由遍历/删除 → 自己用 vector + make_heap（不封装，容器随便用）
vector<int> v;  make_heap(v.begin(), v.end());  push_heap(...);  pop_heap(...);  count(v.begin(), v.end(), 5);
```
// ⚠️ 派生类暴露 protected 容器 c 再 std::count 也能数，但那是 O(n) 且破坏封装（实测 30 万元素 119µs vs 计数表 0µs）

## 3. 算法模板

### 3.1 排序
```cpp
// 选型：通用 sort｜稳定 stable_sort｜只求第 k 小 nth_element(平均 O(n))
//       前 k 个有序 partial_sort｜数逆序对用归并｜值域小用计数（§3.9）
sort(v.begin(), v.end());                          // 升序
sort(v.begin(), v.end(), greater<int>());          // 降序
sort(a + 1, a + n + 1);                            // 数组 1-based
// 结构体排序：
struct Stu { int id, score; };
bool cmp(const Stu& a, const Stu& b) {
    if (a.score != b.score) return a.score > b.score;
    return a.id < b.id;
}
sort(v.begin(), v.end(), cmp);
// 去重三连：
sort(v.begin(), v.end());
v.erase(unique(v.begin(), v.end()), v.end());
// 全排列：
sort(a, a + n);
do { /* 处理排列 */ } while (next_permutation(a, a + n));
// 计数排序（值域小、O(m + 值域)，如 P1271）：编号 1..n 当桶下标
int cnt[N] = {0};
for (int i = 0; i < m; ++i) cin >> tmp, cnt[tmp]++;          // 桶装
for (int i = 1; i <= n; ++i) for (int j = 0; j < cnt[i]; ++j) cout << i << ' ';  // 按桶序倒
// ⚠️ 内层总次数 = Σcnt[i] = m（票数守恒）→ 两段叠加 O(n+m)，不是 O(n×m)；值域大(1e9)开不了数组就别用

// 手写快排（理解分治用；机试直接用 sort）：挖坑法 partition + 递归
void qs(int a[], int l, int r) {                    // 闭区间 [l,r]
    if (l >= r) return;
    int p = a[l], i = l, j = r;                     // 基准挖走 → l 处留坑
    while (i < j) {
        while (i < j && a[j] >= p) j--;  if (i < j) a[i++] = a[j];   // 右找小填左坑
        while (i < j && a[i] <= p) i++;  if (i < j) a[j--] = a[i];   // 左找大填右坑
    }
    a[i] = p;                                       // 碰头处放回基准
    qs(a, l, i - 1);  qs(a, i + 1, r);              // 递归左右
}   // ⚠️ 平均 O(n log n)、最坏 O(n²)（已有序+取首位基准）、不稳定；内层 while 必带 i<j 保护
```

### 3.2 二分
```cpp
// 有序数组中查找
auto it = lower_bound(v.begin(), v.end(), x);      // 第一个 ≥ x
auto it2 = upper_bound(v.begin(), v.end(), x);     // 第一个 > x
// 前驱（最后一个 < x）= *prev(lower_bound(x))；最后一个 ≤ x = *prev(upper_bound(x))
// ⚠️ 两者都要先判 != v.begin()：prev(begin()) 越界 UB；无前驱时即"不存在"（§3.6）
int pos = it - v.begin();                          // 转下标
int cnt = upper_bound(...) - lower_bound(...);     // x 出现次数
if (it != v.end() && *it == x) { /* 真的存在 */ }
// ⚠️ 找不到时返回 end()（x 比所有元素都大）→ 解引用前必须判 it != v.end()
// ⚠️ pos = it - v.begin() 找不到时等于 n：它是合法的"插入位置"，但不能访问 v[pos]
// upper_bound 返回 end() 的条件：x ≥ 最大元素（没有元素 > x）
// v = 1 3 3 3 7：lower_bound(3) → 下标1，upper_bound(3) → 下标4 ⇒ 出现次数 4-1 = 3
// set/map 必须用成员版 s.lower_bound(x)（O(log n)）；全局 lower_bound 对 set 是 O(n)（§2.5）
// 只判存在用 binary_search（返回 bool）；区间必须已按同一比较器有序，否则返回值无保证
// 二分答案框架（check(mid) 判断可行性）
long long l = 0, r = 1e9;
while (l < r) {
    long long mid = (l + r + 1) / 2;
    if (check(mid)) l = mid;
    else r = mid - 1;
}
cout << l;
```

### 3.4 前缀和 / 差分（区间查询与区间加）
```cpp
// 一维前缀和：区间和 [l,r] = pre[r]-pre[l-1]，O(1) 查询（用 1-based，long long）
vector<long long> pre(n + 1, 0);
for (int i = 1; i <= n; i++) { cin >> a[i]; pre[i] = pre[i - 1] + a[i]; }
long long sum = pre[r] - pre[l - 1];

// 一维差分：区间加 [l,r]+k（多次后一次前缀和还原）
vector<long long> diff(n + 2, 0);
auto add = [&](int l, int r, long long k) { diff[l] += k; diff[r + 1] -= k; };
for (int i = 1; i <= n; i++) res[i] = res[i - 1] + diff[i];   // res[i]=位置i增量

long long p2[R + 1][C + 1] = {};
for (int i = 1; i <= R; i++) for (int j = 1; j <= C; j++)
    p2[i][j] = p2[i-1][j] + p2[i][j-1] - p2[i-1][j-1] + a[i][j];
long long sub = p2[x2][y2] - p2[x1-1][y2] - p2[x2][y1-1] + p2[x1-1][y1-1];
// 图解：见 讲义/二维前缀和矩阵图解（实验台 1~16 逐格验证）

// 二维差分：子矩形 (x1,y1)-(x2,y2) 加 k（四角标记 + 两次前缀和还原）
d[x1][y1]+=k; d[x2+1][y1]-=k; d[x1][y2+1]-=k; d[x2+1][y2+1]+=k;
// 还原：for i,j: ans[i][j] = d[i][j] + ans[i-1][j] + ans[i][j-1] - ans[i-1][j-1];
// 差分正向（由最终矩阵 A 求 D，使 A=前缀和(D)）：
//   D[i][j] = A[i][j]-A[i-1][j]-A[i][j-1]+A[i-1][j-1];   // 口诀：自己-上-左+左上，界外=0
```

### 3.7 高精度（加法 / 去前导零，如 P1601）
```cpp
string add(string a, string b){
    reverse(a.begin(), a.end());  reverse(b.begin(), b.end());   // 个位在前好进位
    size_t len = max(a.size(), b.size());
    string ans(len, '0');  int carry = 0;
    for (size_t i = 0; i < len; ++i) {
        int da = i < a.size() ? a[i] - '0' : 0;   // 短的越界补 0（三目，绝不真越界）
        int db = i < b.size() ? b[i] - '0' : 0;
        int t = da + db + carry;
        ans[i] = char('0' + t % 10);
        carry = t / 10;
    }
    if (carry) ans.push_back('1');
    reverse(ans.begin(), ans.end());
    return ans;
}
```

### 3.8 归并排序数逆序对（相邻交换最小次数，如 P1116）

```cpp
long long cnt = 0;  vector<int> a, tmp;
void ms(int l, int r){                      // 闭区间；a[i]<=a[j] 用 <=（相等不成逆序）
    if (l >= r) return;
    int mid = (l + r) >> 1;
    ms(l, mid);  ms(mid + 1, r);
    int i = l, j = mid + 1, k = l;
    while (i <= mid && j <= r)
        if (a[i] <= a[j]) tmp[k++] = a[i++];
        else { cnt += (mid - i + 1); tmp[k++] = a[j++]; }   // 右半先落地→左侧剩余全逆序
    while (i <= mid) tmp[k++] = a[i++];
    while (j <= r)   tmp[k++] = a[j++];
    for (int p = l; p <= r; ++p) a[p] = tmp[p];
}
// ⚠️ 只能交换相邻 → 逆序对；任意交换 → n−环数；别拿快排"跳跃交换"数次数
```

### 4.5 二进制枚举子集（位掩码，如 P1036）
```cpp
// 一个子集 = 一个整数：元素 i(0-based) 在集合里 → 第 i 位为 1
// 空集=0，全集=(1<<n)-1，子集总数 2^n；集合运算：交 &、并 |、补 ^full、包含 (a&b)==a
for (int S = 0; S < (1 << n); ++S) {                    // 枚举所有子集
    if (__builtin_popcount(S) != k) continue;           // 只要恰好 k 个元素的子集
    int sum = 0;
    for (int i = 0; i < n; ++i)
        if (S & (1 << i)) sum += a[i];                  // 第 i 个元素在 S 中
    if (isPrime(sum)) ++ans;
}
// ⚠️ 低位对应第一个元素（二进制串要从右往左读）；n≥31 用 1LL<<n；复杂度 O(2^n · n)，n≤20 可用
// 判断元素在不在：S & (1<<i)；加入/删除：S |= (1<<i) / S &= ~(1<<i)
```

**集合运算 ↔ 位运算对照表**（mask 记作 a、b；合集掩码 `full = (1<<n)-1`，元素 1↔最低位）：

| 想做的事 | 集合说法 | 位运算写法 |
|---|---|---|
| 元素 x 是否在集合中 | x ∈ A | `a >> (x-1) & 1` 或 `a & (1 << (x-1))` |
| 交集 | A ∩ B | `a & b` |
| 并集 | A ∪ B | `a \| b` |
| 补集（全集去掉 A） | ∁A | `a ^ full`（也可 `full & ~a`） |
| 判断 A 是 B 的子集 | A ⊆ B | `(a & b) == a` |
| 判断两个集合相等 | A = B | `a == b` |
| 加入 / 删除元素 x | A ∪ {x} / A \ {x} | `a \|= (1 << k)` / `a &= ~(1 << k)` |
| 集合元素个数 | \|A\| | `__builtin_popcount(a)（返回该数二进制下1的个数）` |

### 4.6 递归下降解析（嵌套结构展开，如 P1928 外星密码）

### 5.1 记忆化搜索（递归 + 缓存：每个状态只算一次）

```cpp
double log(double x);                               // 自然对数 ln x；log10(x)=常用对数，log2(x)=以 2 为底
double pow(double x, double y);                     // x^y（可算多次根：pow(2,0.5)=√2）
double sqrt(double x);                              // √x（sqrt(9)=3）
double fabs(double x);                              // 浮点绝对值（整数用 abs，别用 fabs）
double ceil(double x);                              // 向上取整：≥x 的最小整数（ceil(2.1)=3, ceil(-2.1)=-2）
double floor(double x);                             // 向下取整：≤x 的最大整数（floor(2.9)=2, floor(-2.9)=-3）
double asin(double x);  double acos(double x);      // 反三角，返回【弧度】；定义域 [-1,1]，越界→NaN
double atan(double x);                              // 反正切，定义域 R，值域 (-pi/2,pi/2) ← 丢象限
double atan2(double y, double x);                   // ⭐ 极角，值域 (-pi,pi]；参数顺序 y 在前！
double asinh(double x); double acosh(double x);     // 反双曲（C++11）；acosh 定义域 x>=1
double atanh(double x);                             // 定义域 |x|<1；x=±1 返回 inf（不是 NaN）
// 传参/返回都是 double：整数运算要显式转换，如 (int)floor(a / (double)b)
// ⚠️ 整数幂别用 pow！实测 2..10 的 1..30 次幂 290 组里有 61 组不一致：
//    pow(5,23) 得 11920928955078124（差 1）；pow(5,28) 转 long long 直接变负
//    → 大整数幂用「快速幂 / 循环乘」，浮点幂才用 pow，且用 round() 校正后转整数
// ⚠️ 浮点比较别用 ==（0.1+0.2 != 0.3，sqrt(2)*sqrt(2) = 2.0000000000000004）→ 用 fabs(a-b) < 1e-9
// ⚠️ 整数除法陷阱：5/2 == 2；要浮点结果写 5/2.0 或 (double)5/2
// —— 反三角专项 ——
double ang = atan2(fabs(ax*by-ay*bx), ax*bx+ay*by);  // 两向量夹角 [0,pi]；去 fabs 得带方向角
// ⚠️ 方向角必须用 atan2(y,x)：atan(y/x) 第 2/3 象限全错，且 x=0 时除零
// ⚠️ 夹角别用 acos(dot/(|a||b|))：近平行时 ratio 越界 → acos 给 NaN
//    （实测 200 万组近平行样本中 58.3 万组 NaN）；且 theta<=1e-8 时 acos 直接返回 0
//    （相对误差 100%）；atan2(cross,dot) 全程精确到末位
// ⚠️ 别用 M_PI：-std=c++11/14/17/20 下均未定义（实测），仅 -std=gnu++XX 有 → 写 acos(-1.0)
// ⚠️ 定义域错误只静默返回 NaN（不崩不抛）；NaN 与任何值比较都是 false → 用 isnan() 判
// ⚠️ 极角排序别把 atan2 写进比较器：N=5e5 实测 888.9ms vs 半平面叉积 45.7ms（19.5 倍）
```

### 6.6 类型极值与 INF 哨兵
```cpp
// ① 推荐：<limits>（C++11；T 可变时唯一选择；注意它是函数，要带 ()）
numeric_limits<int>::max();          // 2147483647
numeric_limits<int>::min();          // -2147483648（整数 = 最负值）
numeric_limits<long long>::max();    // 9223372036854775807
numeric_limits<unsigned>::min();     // 0（无符号！）
numeric_limits<double>::max();       // 1.797693e+308
// ② C 风格宏（打得快）：<climits>/<cfloat>
INT_MAX  INT_MIN  LLONG_MAX  LLONG_MIN  UINT_MAX  ULLONG_MAX  DBL_MAX  DBL_EPSILON
// ③ 竞赛 INF 哨兵（0x3f 技巧）：memset 0x3f → 1061109567，两倍相加仍 < INT_MAX，不会溢出
const int INF = 0x3f3f3f3f;   int dp[N];  memset(dp, 0x3f, sizeof(dp));
// ⚠️【最易错】浮点 min() 是"最小正数"，不是最负值！取最负要用 lowest()
//    numeric_limits<double>::min()    = 2.225074e-308  ← 最小正规格化数
//    numeric_limits<double>::lowest() = -1.797693e+308 ← 真正的最负值 ✔
// ⚠️ 其它溢出坑：unsigned max+1 回绕成 0；INT_MIN/-1、abs(INT_MIN) 都溢出（UB）；
//    用 INT_MAX 当 INF 再相加会溢出 → 用 0x3f3f3f3f 或 long long
// ⚠️【取值对了但打印错】char/unsigned char 的极值必须 (int) 强转：
//    cout << numeric_limits<char>::max() 打印的是字符，不是数字
//    五种修法（输出 127）：(int)c ／ +c（一元加号整型提升，最短）／ static_cast<int>(c)
//    ／ printf("%d", c) ／ to_string((int)c)
//    int8_t/uint8_t 就是 signed char/unsigned char 的别名 → 同样要强转；bool 不受影响
// ⚠️ printf 类型必须配对：%d↔int、%lld↔long long（配错可能打印 -1 或"碰巧正确"的 UB）；加 -Wall -Wformat 会直接报警告
```

### 6.7 球面距离（Haversine，经纬度两点）
```cpp
const double PI = acos(-1.0), R = 6371.0;          // 地球半径 km
double hav(double x){ double s = sin(x/2); return s*s; }  // = (1-cos x)/2，但无相消
double lat1, lon1, lat2, lon2; cin >> lat1 >> lon1 >> lat2 >> lon2;
lat1 *= PI/180; lon1 *= PI/180; lat2 *= PI/180; lon2 *= PI/180;   // ★ 角度→弧度
double h = hav(lat2-lat1) + cos(lat1)*cos(lat2)*hav(lon2-lon1);
if (h > 1) h = 1;  if (h < 0) h = 0;               // 兜底，防 sqrt/asin 越界
printf("%.4fkm\n", R * 2 * asin(sqrt(h)));         // 圆心角 = 2*asin(sqrt(h))
// ⚠️ 输入是【角度】，sin/cos 吃【弧度】→ 不转就是 WA（实测 8775.07 vs 正确 5793.22，差 51.5%）
// ⚠️ hav 别写 (1-cos x)/2：小角度灾难性相消，x<=1e-8 时 h 直接变 0
// ⚠️ 反解别写 acos(1-2h)：第二次相消，h 小时 1-2h 舍入成 1 → acos(1)=0
//    实测 theta=1e-8 时三种旧写法全输出 0，只有 sin^2/2 + 2asin(sqrt) 正确
// ⚠️ 自测三连：同点→0；对径点(0,0)-(0,180)→20015.0868=pi*R；南北极→20015.0868
```

### 7.1 fgets 读行 / sscanf 解析 / snprintf 拼装（字符串三件套）
```cpp
char buf[256];
fgets(buf, sizeof(buf), stdin);     // 读整行（保留 '\n'）
buf[strcspn(buf, "\n")] = 0;        // 去末尾 '\n'（strcspn 找第一个 '\n'）

int a; char name[20]; double x;
sscanf(buf, "%d %s %lf", &a, name, &x);   // 从字符串解析（数组名不用 &）
if (sscanf(buf, "%d %s", &a, name) != 2) { /* 解析失败，别用垃圾值 */ }
sscanf(buf, "%d %[^\n]", &a, name);        // 读含空格的字段用 %[^\n]

char out[256];
sprintf(out, "a=%d name=%s %.2f", a, name, x);   // 拼装
snprintf(out, sizeof(out), "a=%d", a);           // 更安全：限定总大小防溢出
```
